# tests/test_fribourg_parser.py
import pytest
from datetime import datetime
from scrapy.http import HtmlResponse, Request
from tribuna_scraper.spiders.fribourg import FribourgSpider
from tribuna_scraper.utils.normalizers import TribunaNormalizer

class TestFribourgParser:
    """Tests unitaires pour le parser Fribourg"""
    
    @pytest.fixture
    def spider(self):
        return FribourgSpider()
    
    @pytest.fixture
    def normalizer(self):
        return TribunaNormalizer()
    
    @pytest.fixture
    def sample_html(self):
        """HTML d'exemple simplifié de Fribourg"""
        return """
        <html>
        <body>
            <table>
                <tr>
                    <td>04.06.2025</td>
                    <td>101_2025_26</td>
                    <td>IIe Cour d'appel civil</td>
                    <td>
                        <a href="/tribunavtplus/ServletDownload/101_2025_26_a0f56a97c2d648c39aa521b0f661ff9f.pdf?dossiernummer=101_2025_26&path=xyz&pathIsEncrypted=1">
                            Arrêt du 4 juin 2025
                        </a>
                    </td>
                </tr>
            </table>
        </body>
        </html>
        """
    
    @pytest.fixture
    def mock_response(self, sample_html):
        """Crée une réponse Scrapy mockée"""
        request = Request(url='https://publicationtc.fr.ch/?locale=fr')
        response = HtmlResponse(
            url='https://publicationtc.fr.ch/?locale=fr',
            request=request,
            body=sample_html.encode('utf-8'),
            encoding='utf-8'
        )
        return response
    
    def test_normalize_date(self, normalizer):
        """Test la normalisation des dates"""
        assert normalizer.normalize_date('04.06.2025') == '2025-06-04'
        assert normalizer.normalize_date('31/12/2024') == '2024-12-31'
        assert normalizer.normalize_date('2024-01-15') == '2024-01-15'
        assert normalizer.normalize_date('invalid') is None
    
    def test_extract_docket_number(self, normalizer):
        """Test l'extraction des numéros de dossier"""
        assert normalizer.extract_docket_number('101_2025_26') == '101_2025_26'
        assert normalizer.extract_docket_number('Arrêt 2C_123/2024') == '2C_123/2024'
        assert normalizer.extract_docket_number('CO 456/2023 décision') == 'CO 456/2023'
    
    def test_parse_servlet_url(self, normalizer):
        """Test le parsing des URLs ServletDownload"""
        url = 'https://site.ch/ServletDownload/123_abc.pdf?dossiernummer=123&path=xyz&pathIsEncrypted=1'
        params = normalizer.parse_servlet_url(url)
        
        assert params['pdf_id'] == '123_abc'
        assert params['dossiernummer'] == '123'
        assert params['path'] == 'xyz'
        assert params['pathIsEncrypted'] is True
    
    def test_parse_decision_from_table(self, spider, mock_response):
        """Test l'extraction d'une décision depuis une table"""
        items = list(spider.parse_decisions_list(mock_response))
        
        assert len(items) == 1
        item = items[0]
        
        assert item['decision_date'] == '2025-06-04'
        assert item['docket_number'] == '101_2025_26'
        assert item['court'] == 'IIe Cour d\'appel civil'
        assert item['title'] == 'Arrêt du 4 juin 2025'
        assert 'ServletDownload' in item['pdf_url']
        assert item['source'] == 'publicationtc.fr.ch'
        assert item['locale'] == 'fr'
    
    def test_guess_locale(self, normalizer):
        """Test la détection de la langue"""
        assert normalizer.guess_locale('https://site.fr.ch/?locale=de') == 'de'
        assert normalizer.guess_locale('https://site.ch', 'Tribunal cantonal') == 'fr'
        assert normalizer.guess_locale('https://site.ch', 'Kantonsgericht') == 'de'
        assert normalizer.guess_locale('https://site.ch', 'Tribunale cantonale') == 'it'
